def drain_pending(queue, sink, retries=3):
    """Flush queued events while retaining failed work for a later retry."""
    pending = list(queue)
    queue.clear()
    for event in pending:
        for attempt in range(retries):
            try:
                sink.write(event)
                break
            except OSError:
                if attempt + 1 == retries:
                    queue.append(event)
    return len(pending) - len(queue)
